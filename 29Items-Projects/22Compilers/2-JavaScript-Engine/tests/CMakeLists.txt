function(jsengine_add_test test_name source_file)
    add_executable(${test_name} ${source_file})
    target_link_libraries(${test_name} PRIVATE jsengine::jsengine)
    jsengine_apply_default_options(${test_name})
    add_test(NAME ${test_name} COMMAND ${test_name})
endfunction()
